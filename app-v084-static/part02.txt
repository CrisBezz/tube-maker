function updateSymmetryPlane() {
  if (symmetryPlaneHelper) {
    symmetryPlaneHelper.geometry.dispose();
    symmetryPlaneHelper.material.dispose();
    symmetryPlaneHelper.removeFromParent();
    symmetryPlaneHelper = null;
  }
  if (!sourceRoot || !symmetry.checked || !showSymmetryPlane.checked) return;
  const box = sourceBounds();
  if (box.isEmpty()) return;
  const size = box.getSize(new THREE.Vector3());
  const centre = box.getCenter(new THREE.Vector3());
  const axis = symmetryAxis.value;
  let width, height;
  if (axis === 'x') { width = Math.max(size.z, 0.001); height = Math.max(size.y, 0.001); }
  else if (axis === 'y') { width = Math.max(size.x, 0.001); height = Math.max(size.z, 0.001); }
  else { width = Math.max(size.x, 0.001); height = Math.max(size.y, 0.001); }
  const geo = new THREE.PlaneGeometry(width * 1.15, height * 1.15);
  if (axis === 'x') geo.rotateY(Math.PI / 2);
  else if (axis === 'y') geo.rotateX(-Math.PI / 2);
  const mat = new THREE.MeshBasicMaterial({ color: 0x55b7ff, transparent: true, opacity: 0.18, side: THREE.DoubleSide, depthWrite: false });
  symmetryPlaneHelper = new THREE.Mesh(geo, mat);
  symmetryPlaneHelper.position.copy(centre);
  symmetryPlaneHelper.renderOrder = 10;
  scene.add(symmetryPlaneHelper);
}
function disposeObject(root) {
  if (!root) return;
  root.traverse((obj) => {
    if (obj.geometry) obj.geometry.dispose?.();
    const mat = obj.material;
    if (Array.isArray(mat)) mat.forEach((m) => m.dispose?.()); else mat?.dispose?.();
  });
  root.removeFromParent();
}
function fitCamera(object) {
  const box = new THREE.Box3().setFromObject(object);
  const sphere = box.getBoundingSphere(new THREE.Sphere());
  if (!Number.isFinite(sphere.radius) || sphere.radius <= 0) return;
  controls.target.copy(sphere.center);
  camera.near = Math.max(sphere.radius / 1000, 0.001);
  camera.far = Math.max(sphere.radius * 100, 100);
  camera.position.copy(sphere.center).add(new THREE.Vector3(1.5, 1.1, 1.5).normalize().multiplyScalar(sphere.radius * 3));
  camera.updateProjectionMatrix();
  controls.update();
  grid.scale.setScalar(Math.max(sphere.radius, 1));
  grid.position.y = box.min.y;
}
function normaliseImported(root) {
  const group = new THREE.Group();
  root.updateMatrixWorld(true);
  root.traverse((obj) => {
    if (!obj.isMesh || !obj.geometry) return;
    const geometry = obj.geometry.clone();
    geometry.applyMatrix4(obj.matrixWorld);
    if (!geometry.attributes.normal) geometry.computeVertexNormals();
    const material = new THREE.MeshStandardMaterial({ color: 0x9ba5b4, roughness: 0.75, metalness: 0.05, side: THREE.DoubleSide, transparent: true, opacity: 0.42 });
    group.add(new THREE.Mesh(geometry, material));
  });
  return group;
}
function parseObjData(text) {
  const vertices = [];
  const edgeMap = new Map();
  const faces = [];
  const lines = text.split(/\r?\n/);
  const resolveIndex = (raw) => {
    const value = Number(raw);
    return value < 0 ? vertices.length + value : value - 1;
  };
  const addIndexEdge = (ia, ib) => {
    if (ia == null || ib == null || !vertices[ia] || !vertices[ib]) return;
    const a = vertices[ia], b = vertices[ib];
    const key = ia < ib ? `${ia}|${ib}` : `${ib}|${ia}`;
    if (!edgeMap.has(key)) edgeMap.set(key, [a.clone(), b.clone()]);
  };
  for (const rawLine of lines) {
    const line = rawLine.trim();
    if (!line || line.startsWith('#')) continue;
    const parts = line.split(/\s+/);
    if (parts[0] === 'v' && parts.length >= 4) {
      vertices.push(new THREE.Vector3(Number(parts[1]), Number(parts[2]), Number(parts[3])));
    }
    if (parts[0] === 'f' && parts.length >= 4) {
      const ids = parts.slice(1).map((token) => resolveIndex(token.split('/')[0])).filter((id) => id != null);
      if (ids.length === 4) {
        faces.push(ids.map((id) => vertices[id].clone()));
        for (let i = 0; i < 4; i++) addIndexEdge(ids[i], ids[(i + 1) % 4]);
      }
    }
  }
  return { edges: [...edgeMap.values()], faces };
}
async function loadFile(file) {
  setStatus(`Loading ${file.name}…`);
  sourceName = file.name.replace(/\.[^.]+$/, '') || 'mesh-line-replacer';
  originalObjQuadEdges = null;
  originalObjQuadFaces = null;
  const ext = file.name.split('.').pop()?.toLowerCase();
  let loaded;
  if (ext === 'obj') {
    const text = await file.text();
    const parsed = parseObjData(text);
    originalObjQuadEdges = parsed.edges;
    originalObjQuadFaces = parsed.faces;
    loaded = new OBJLoader().parse(text);
  } else {
    const url = URL.createObjectURL(file);
    try {
      if (ext === 'glb' || ext === 'gltf') loaded = (await new GLTFLoader().loadAsync(url)).scene;
      else if (ext === 'stl') loaded = new THREE.Mesh(await new STLLoader().loadAsync(url));
      else throw new Error('Unsupported file format.');
    } finally { URL.revokeObjectURL(url); }
  }
  setSource(normaliseImported(loaded));
  if (ext === 'obj' && originalObjQuadEdges?.length) edgeMode.value = 'objQuads';
  else if (edgeMode.value === 'objQuads') edgeMode.value = 'quads';
  setStatus(`${file.name} loaded${originalObjQuadEdges?.length ? ` • ${originalObjQuadEdges.length.toLocaleString()} original quad edges` : ''}`);
}
