
// v0.9.0-wip.9 — independent faceted shading VIEW option
(function(){
const VERSION='0.9.0-wip.9';
document.title=`Mesh Line Replacer v${VERSION}`;
const vv=document.querySelector('.version');if(vv)vv.textContent=`v${VERSION}`;
const av=document.querySelector('.about-version');if(av)av.textContent=`Version ${VERSION}`;

const viewBtn=document.getElementById('mlrView');
const viewPop=document.querySelector('.mlr-viewpop2');
const viewList=viewPop?.querySelector('#mlrViewList');
const base=document.getElementById('baseViewMode')||document.getElementById('viewMode');
if(!viewBtn||!viewPop||!viewList||!base)return;

let faceted=document.getElementById('viewFacetedShading');
if(!faceted){faceted=document.createElement('input');faceted.type='checkbox';faceted.id='viewFacetedShading';faceted.checked=false;}
const originalFlat=new WeakMap();
function materialsOf(mat){return Array.isArray(mat)?mat:[mat]}
function roots(){const arr=[];try{if(typeof sourceRoot!=='undefined'&&sourceRoot)arr.push(sourceRoot)}catch{}try{if(typeof resultRoot!=='undefined'&&resultRoot)arr.push(resultRoot)}catch{}return arr}
function applyFaceted(){
 const on=faceted.checked;
 for(const root of roots())root.traverse?.(obj=>{
   if(!obj.isMesh&&!obj.isInstancedMesh)return;
   for(const mat of materialsOf(obj.material)){
     if(!mat||!('flatShading' in mat))continue;
     if(!originalFlat.has(mat))originalFlat.set(mat,!!mat.flatShading);
     mat.flatShading=on?true:originalFlat.get(mat);
     mat.needsUpdate=true;
   }
 });
}
faceted.addEventListener('change',applyFaceted);

function anchor(pop,btn){const r=btn.getBoundingClientRect();pop.style.left=Math.max(10,Math.min(r.left,window.innerWidth-pop.offsetWidth-10))+'px';pop.style.top=Math.max(r.bottom+6,54)+'px'}
function closeOthers(){document.querySelectorAll('.mlr-filepop,.mlr-modepop').forEach(p=>p.hidden=true)}
function refreshActive(){viewList.querySelectorAll('button[data-view-value]').forEach(b=>b.classList.toggle('is-active',b.dataset.viewValue===base.value))}
function menuCheck(input,label){const row=document.createElement('label');row.className='mlr-menurow mlr-check';const span=document.createElement('span');span.textContent=label;row.append(span,input);return row}
function rebuild(){
 viewList.innerHTML='';
 for(const opt of [...base.options]){
   const b=document.createElement('button');b.type='button';b.dataset.viewValue=opt.value;b.textContent=opt.textContent.trim();b.classList.toggle('is-active',opt.value===base.value);
   b.onclick=()=>{base.value=opt.value;base.dispatchEvent(new Event('change',{bubbles:true}));refreshActive();setTimeout(applyFaceted,0);viewPop.hidden=true};viewList.appendChild(b);
 }
 const target=document.getElementById('viewTarget');
 if(target){viewList.appendChild(document.createElement('hr'));const row=document.createElement('label');row.className='mlr-menurow';const span=document.createElement('span');span.textContent='View target';row.append(span,target);viewList.appendChild(row)}
 viewList.appendChild(document.createElement('hr'));
 viewList.appendChild(menuCheck(faceted,'Faceted shading'));
 const overlays=[['overlayWire','Wireframe overlay'],['overlayFaceNormals','Face normals'],['overlayVertexNormals','Vertex normals']];
 for(const [id,label] of overlays){const input=document.getElementById(id);if(input)viewList.appendChild(menuCheck(input,label))}
 const length=document.getElementById('normalLength');if(length){const row=document.createElement('label');row.className='mlr-menurow';const span=document.createElement('span');span.textContent='Normal length';row.append(span,length);viewList.appendChild(row)}
}

const replacement=viewBtn.cloneNode(true);viewBtn.replaceWith(replacement);
replacement.addEventListener('click',e=>{e.preventDefault();e.stopPropagation();const opening=viewPop.hidden;closeOthers();if(opening){rebuild();viewPop.hidden=false;requestAnimationFrame(()=>anchor(viewPop,replacement))}else viewPop.hidden=true});
base.addEventListener('change',()=>{refreshActive();setTimeout(applyFaceted,0)});
document.addEventListener('pointerdown',e=>{if(!viewPop.hidden&&!viewPop.contains(e.target)&&e.target!==replacement)viewPop.hidden=true},{capture:true});
window.addEventListener('resize',()=>{if(!viewPop.hidden)anchor(viewPop,replacement)});
for(const id of ['generateBtn','generate','buildBtn'])document.getElementById(id)?.addEventListener('click',()=>setTimeout(applyFaceted,60));
const mode=document.getElementById('replaceMode')||document.getElementById('mlrMode');mode?.addEventListener('change',()=>setTimeout(applyFaceted,30));
rebuild();applyFaceted();
})();
