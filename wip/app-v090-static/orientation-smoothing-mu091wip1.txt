let cachedFaces=null,cachedSignature='';
function faceCentre(face){return face.reduce((a,p)=>a.add(p),new THREE.Vector3()).multiplyScalar(1/Math.max(face.length,1))}
function collectContext(){
 try{
   const result=typeof collectFaces==='function'?collectFaces():null;let faces=result?.faces||[];
   if(!faces.length)return[];
   if(typeof orientTorusFaces==='function')faces=orientTorusFaces(faces);
   const sig=`${faces.length}:${faces[0]?.length||0}:${faces[0]?.[0]?.x||0}`;
   if(sig===cachedSignature&&cachedFaces)return cachedFaces;
   cachedSignature=sig;
   cachedFaces=faces.map(face=>{const centre=faceCentre(face),normal=typeof polygonNormal==='function'?polygonNormal(face):null;if(!normal)return null;const frame=nativeFrame(face,normal.clone(),centre.clone());return frame?{face,centre,normal:frame.normal.clone().normalize(),xAxis:frame.xAxis.clone().normalize()}:null}).filter(Boolean);
   return cachedFaces;
 }catch(e){console.warn('MeshUtilz smoothing context failed',e);return[]}
}
function strength(){const m=currentMode(),p=m==='custom'?'custom':'torus';const field=document.getElementById(p+'OrientationField');if(field?.value!=='smooth')return 0;return Math.max(0,Math.min(1,Number(document.getElementById(p+'OrientationSmooth')?.value||0)/100))}
function smoothFrame(base,point){
 const amount=strength();if(amount<=0)return base;
 const context=collectContext();if(context.length<2)return base;
 // Find a small spatial neighbourhood. Limiting to 12 keeps this prototype responsive on denser meshes.
 const near=context.map(r=>({r,d:r.centre.distanceToSquared(point)})).sort((a,b)=>a.d-b.d).slice(0,12);
 const baseN=base.normal.clone().normalize(),baseX=base.xAxis.clone().normalize();
 const avgN=new THREE.Vector3(),avgX=new THREE.Vector3();let wsum=0;
 for(const {r,d} of near){
   const w=1/(Math.sqrt(d)+1e-6);const n=r.normal.clone();if(n.dot(baseN)<0)n.negate();let x=r.xAxis.clone();if(x.dot(baseX)<0)x.negate();avgN.addScaledVector(n,w);avgX.addScaledVector(x,w);wsum+=w;
 }
 if(wsum<=0||avgN.lengthSq()<1e-12)return base;
 avgN.normalize();
 const n=baseN.clone().lerp(avgN,amount).normalize();
 // Tangent averaging removes rotational chatter. Project back into the smoothed normal plane.
 avgX.multiplyScalar(1/wsum);avgX.addScaledVector(n,-avgX.dot(n));if(avgX.lengthSq()<1e-12){avgX.copy(baseX).addScaledVector(n,-baseX.dot(n))}avgX.normalize();
 let x=baseX.clone();x.addScaledVector(n,-x.dot(n));if(x.lengthSq()<1e-12)x.copy(avgX);else x.normalize();x.lerp(avgX,amount);x.addScaledVector(n,-x.dot(n));if(x.lengthSq()<1e-12)x.copy(avgX);x.normalize();
 const y=n.clone().cross(x).normalize();
 return{xAxis:x,yAxis:y,normal:n};
}
// Wrap the existing frame builders, leaving placement positions, scaling, tilt and axis correction untouched.

torusNodeFrame=function(node){const base=nativeNodeFrame(node);if(!base||strength()<=0||!node?.point)return base;return smoothFrame(base,node.point)};
