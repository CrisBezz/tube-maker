function txt(el){return(el?.textContent||'').replace(/\s+/g,' ').trim().toLowerCase()}
function ids(el){return[...el.querySelectorAll?.('[id]')||[]].map(x=>x.id).join(' ').toLowerCase()}
function visibleControls(sec){return[...sec.querySelectorAll('.control,label.check')].filter(c=>{const s=getComputedStyle(c);return s.display!=='none'&&!c.classList.contains('mlr10-hidden')&&!c.classList.contains('mlr11-hidden')&&!c.classList.contains('mlr-consolidated-hidden')&&!c.classList.contains('mlr-wip6-hidden')})}
function pairWithin(sec,tests){const body=sec.querySelector('.mlr10-body');if(!body)return;for(const [aRx,bRx] of tests){const cs=[...body.children].filter(x=>x.classList?.contains('control')&&!x.closest('.mlr11-pair,.mlr10-pair'));const a=cs.find(c=>aRx.test(txt(c)+' '+ids(c)));const b=cs.find(c=>c!==a&&bRx.test(txt(c)+' '+ids(c)));if(a&&b){const row=document.createElement('div');row.className='mlr11-pair';body.insertBefore(row,a);row.append(a,b)}}}
function collapseSingleControlHeadings(){for(const sec of panel.querySelectorAll('.mlr10-section')){const cs=visibleControls(sec);if(cs.length===1&&sec.dataset.mlr10!=='Geometry / Quality'){sec.classList.add('mlr11-inline');sec.open=true}else sec.classList.remove('mlr11-inline')}}
function activeModeText(){const e=document.getElementById('replaceMode')||document.getElementById('mlrMode');return((e?.value||'')+' '+(e?.selectedOptions?.[0]?.textContent||'')).toLowerCase()}
function context(){const mode=activeModeText();

 // Size mode: show only the active branch.
 for(const prefix of ['torus','custom']){const sm=document.getElementById(prefix+'SizeMode');if(!sm)continue;const constant=sm.value==='constant';for(const c of panel.querySelectorAll('.control')){const key=ids(c)+' '+txt(c);if(!key.includes(prefix))continue;const isConst=/constant/.test(key);const isRel=/relative|scale/.test(key)&&!isConst;if(isConst)c.classList.toggle('mlr11-hidden',!constant);if(isRel)c.classList.toggle('mlr11-hidden',constant)}}
 // Placement mode: hide node-only helpers unless node placement is active.
 for(const prefix of ['torus','custom']){const pm=document.getElementById(prefix+'PlacementMode');if(!pm)continue;const node=pm.value==='node';for(const c of panel.querySelectorAll('.control')){const key=ids(c)+' '+txt(c);if(!key.includes(prefix))continue;if(/node normal|averaged normal|shared.*node|shared.*vertex/.test(key)&&!key.includes('placementmode'))c.classList.toggle('mlr11-hidden',!node)}}
 for(const sec of panel.querySelectorAll('.mlr10-section')){const cs=visibleControls(sec);sec.classList.toggle('mlr11-hidden',cs.length===0);if(sec.dataset.mlr10==='Geometry / Quality')sec.open=false}
 collapseSingleControlHeadings();
}

for(const id of ['replaceMode','mlrMode','torusSizeMode','customSizeMode','torusPlacementMode','customPlacementMode'])document.getElementById(id)?.addEventListener('change',()=>requestAnimationFrame(context));
