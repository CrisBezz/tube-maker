function parseGlbPrimitiveGroups(file){
  if(!file||!file.name?.toLowerCase().endsWith('.glb')){glbPrimitiveGroups=[];return;}
  file.arrayBuffer().then(buf=>{
    try{
      const dv=new DataView(buf); if(dv.getUint32(0,true)!==0x46546c67)return;
      let off=12,json=null;
      while(off+8<=buf.byteLength){const len=dv.getUint32(off,true),type=dv.getUint32(off+4,true);off+=8;if(type===0x4e4f534a){json=JSON.parse(new TextDecoder().decode(new Uint8Array(buf,off,len)).replace(/\0+$/,''));break}off+=len;}
      const groups=[];for(const mesh of json?.meshes||[])for(const prim of mesh.primitives||[])groups.push(prim?.extras?.nomad?.group ?? null);
      glbPrimitiveGroups=groups;
    }catch(e){console.warn('MeshUtilz facegroup metadata parse failed',e);glbPrimitiveGroups=[];}
  });
}

function qkey(p,t){return `${Math.round(p.x/t)},${Math.round(p.y/t)},${Math.round(p.z/t)}`}
function sourceBounds(){const b=new THREE.Box3();try{sourceRoot?.updateMatrixWorld?.(true);sourceRoot?.traverse?.(o=>{if(o.isMesh&&o.geometry?.attributes?.position)b.expandByObject(o)})}catch{}return b}
function collectGroupedTriangles(){
 if(!sourceRoot)return[];sourceRoot.updateMatrixWorld(true);const tris=[];let primitiveIndex=0;
 sourceRoot.traverse(obj=>{
   if(!obj.isMesh||!obj.geometry?.attributes?.position)return;const g=obj.geometry,p=g.getAttribute('position'),idx=g.index;const groups=g.groups?.length?g.groups:[{start:0,count:idx?idx.count:p.count}];
   for(let gi=0;gi<groups.length;gi++){
     const gr=groups[gi],fallback=glbPrimitiveGroups[primitiveIndex] ?? glbPrimitiveGroups[primitiveIndex+gi] ?? primitiveIndex+gi;const gid=groupFromObject(obj,fallback);
     const start=gr.start||0,end=Math.min(start+(gr.count||0),idx?idx.count:p.count);
     for(let i=start;i+2<end;i+=3){const ia=idx?idx.getX(i):i,ib=idx?idx.getX(i+1):i+1,ic=idx?idx.getX(i+2):i+2;const a=new THREE.Vector3(p.getX(ia),p.getY(ia),p.getZ(ia)).applyMatrix4(obj.matrixWorld),b=new THREE.Vector3(p.getX(ib),p.getY(ib),p.getZ(ib)).applyMatrix4(obj.matrixWorld),c=new THREE.Vector3(p.getX(ic),p.getY(ic),p.getZ(ic)).applyMatrix4(obj.matrixWorld);const n=b.clone().sub(a).cross(c.clone().sub(a));if(n.lengthSq()>1e-18)tris.push({a,b,c,normal:n.normalize(),group:gid});}
   }
   primitiveIndex+=Math.max(groups.length,1);
 });return tris;
}
function buildBoundarySegments(){
 const tris=collectGroupedTriangles();if(!tris.length)return[];const box=sourceBounds(),diag=Math.max(box.getSize(new THREE.Vector3()).length(),1e-6),tol=diag*1e-6;const edges=new Map();
 const add=(a,b,tri)=>{const ka=qkey(a,tol),kb=qkey(b,tol),key=ka<kb?`${ka}|${kb}`:`${kb}|${ka}`;let e=edges.get(key);if(!e){e={ka:ka<kb?ka:kb,kb:ka<kb?kb:ka,a:(ka<kb?a:b).clone(),b:(ka<kb?b:a).clone(),refs:[]};edges.set(key,e)}e.refs.push(tri)};
 for(const t of tris){add(t.a,t.b,t);add(t.b,t.c,t);add(t.c,t.a,t)}
 const centre=box.getCenter(new THREE.Vector3()),segments=[];const normalMode=activeMode()==='custom'?document.getElementById('customNormalMode')?.value:document.getElementById('torusNormalMode')?.value;
 for(const e of edges.values()){
   const groups=new Set(e.refs.map(r=>String(r.group)));if(groups.size<2)continue;
   let n=new THREE.Vector3(),seed=e.refs[0].normal.clone();for(const r of e.refs){const rn=r.normal.clone();if(rn.dot(seed)<0)rn.negate();n.add(rn)}if(n.lengthSq()<1e-14)continue;n.normalize();const mid=e.a.clone().add(e.b).multiplyScalar(.5),radial=mid.clone().sub(centre);if(normalMode==='outward'&&n.dot(radial)<0)n.negate();if(normalMode==='inward'&&n.dot(radial)>0)n.negate();segments.push({...e,normal:n});
 }
 return segments;
}
function traceBoundaryChains(segments){
 const adj=new Map();segments.forEach((s,i)=>{for(const k of [s.ka,s.kb]){let a=adj.get(k);if(!a){a=[];adj.set(k,a)}a.push(i)}});const used=new Uint8Array(segments.length),chains=[];
 function walk(startEdge,startKey){const points=[],normals=[];let ei=startEdge,key=startKey,guard=0;while(ei!=null&&!used[ei]&&guard++<segments.length+2){const s=segments[ei];used[ei]=1;const forward=s.ka===key,from=forward?s.a:s.b,to=forward?s.b:s.a,nextKey=forward?s.kb:s.ka;if(!points.length)points.push(from.clone());points.push(to.clone());normals.push(s.normal.clone());const next=(adj.get(nextKey)||[]).find(j=>!used[j]);if(next==null)break;key=nextKey;ei=next}return{points,normals,closed:points.length>2&&points[0].distanceToSquared(points[points.length-1])<1e-12}}
 for(const [key,list] of adj)if(list.length!==2)for(const ei of list)if(!used[ei])chains.push(walk(ei,key));for(let i=0;i<segments.length;i++)if(!used[i])chains.push(walk(i,segments[i].ka));return chains.filter(c=>c.points.length>1);
}
function sampleChain(chain,wanted){
 const pts=chain.points,segLen=[];let total=0;for(let i=0;i<pts.length-1;i++){const l=pts[i].distanceTo(pts[i+1]);segLen.push(l);total+=l}if(total<=1e-9)return[];const closed=chain.closed;let count=closed?Math.max(1,Math.round(total/wanted)):Math.max(2,Math.floor(total/wanted)+1);const step=closed?total/count:total/(count-1),out=[];
 for(let si=0;si<count;si++){let d=closed?si*step:Math.min(total,si*step),acc=0,idx=0;while(idx<segLen.length-1&&acc+segLen[idx]<d){acc+=segLen[idx];idx++}const l=Math.max(segLen[idx],1e-9),u=Math.max(0,Math.min(1,(d-acc)/l)),p=pts[idx].clone().lerp(pts[idx+1],u),tangent=pts[idx+1].clone().sub(pts[idx]).normalize(),normal=(chain.normals[idx]||chain.normals[Math.max(0,idx-1)]||new THREE.Vector3(0,1,0)).clone().normalize();out.push({point:p,normal,tangent,localRadius:step*.5,isFacegroupBoundary:true,surfaceNormal:normal.clone(),boundaryTangent:tangent.clone()})}return out;
}
