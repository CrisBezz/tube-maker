
// v0.9.0-wip.6 — final scroll-panel de-duplication pass
(function(){

const vv=document.querySelector('.version');if(vv)vv.textContent=`v${VERSION}`;

const panel=document.querySelector('.panel,.sidebar,.controls');
const viewPop=document.querySelector('.mlr-viewpop2');

}

if(panel){for(const d of panel.querySelectorAll('.mlr-section')){const body=d.querySelector('.mlr-section-body');if(!body)continue;const visible=[...body.children].filter(x=>!x.classList.contains('mlr-wip6-hidden')&&!x.classList.contains('mlr-consolidated-hidden')&&getComputedStyle(x).display!=='none');if(!visible.length)hide(d)}}
