
// MeshUtilz v0.9.2-wip.21 — export sync fix: capture generated matrices at export if WIP20 callback missed them
(function(){
function extractNow(){try{if(typeof resultRoot==='undefined'||!resultRoot)return[];resultRoot.updateMatrixWorld?.(true);const sets=[];resultRoot.traverse?.(o=>{if(o?.isInstancedMesh&&o.count>0)sets.push(o)});if(!sets.length)return[];sets.sort((a,b)=>b.count-a.count);const src=sets[0],m=new THREE.Matrix4(),wm=new THREE.Matrix4(),p=new THREE.Vector3(),q=new THREE.Quaternion(),s=new THREE.Vector3(),out=[];for(let i=0;i<src.count;i++){src.getMatrixAt(i,m);wm.multiplyMatrices(src.matrixWorld||new THREE.Matrix4(),m);wm.decompose(p,q,s);if(Math.abs(s.x)<1e-10||Math.abs(s.y)<1e-10||Math.abs(s.z)<1e-10)continue;out.push({position:p.clone(),quaternion:q.clone(),scale:s.clone(),matrix:wm.toArray()})}return out}catch(e){console.warn('WIP21 export sync extraction failed',e);return[]}}
