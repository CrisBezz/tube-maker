
// MeshUtilz v0.9.2-wip.17 — make Live NOM preview use same fallback placement source as exporter
(function(){
const VERSION='0.9.2-wip.17',PRODUCT='MeshUtilz';
function stamp(){document.title=`${PRODUCT} v${VERSION}`;document.querySelectorAll('.version').forEach(x=>x.textContent=`v${VERSION}`);const av=document.querySelector('.about-version');if(av)av.textContent=`${PRODUCT} · Version ${VERSION}`;const bars=[...document.querySelectorAll('.mlr-statusbar,.mlr-wip8-statusbar')];bars.slice(1).forEach(x=>x.remove());if(bars[0]){const s=bars[0].querySelectorAll('span');if(s.length)s[s.length-1].textContent=`v${VERSION}`}}
// Publish deterministic face-centre placements so WIP16 can consume them outside the older private module scope.
function publish(){try{if(typeof sourceRoot==='undefined'||!sourceRoot)return 0;sourceRoot.updateMatrixWorld(true);const out=[];sourceRoot.traverse(o=>{if(!o.isMesh||!o.geometry?.attributes?.position)return;const p=o.geometry.getAttribute('position'),ix=o.geometry.index;const count=ix?ix.count:p.count;for(let i=0;i+2<count;i+=3){const ids=[ix?ix.getX(i):i,ix?ix.getX(i+1):i+1,ix?ix.getX(i+2):i+2],c=new THREE.Vector3();ids.forEach(k=>c.add(new THREE.Vector3(p.getX(k),p.getY(k),p.getZ(k))));c.multiplyScalar(1/3).applyMatrix4(o.matrixWorld);out.push({position:c})}});window.__muLivePreviewPlacements=out;return out.length}catch{return 0}}
function refresh(){const n=publish();try{if(n&&typeof rebuild==='function')rebuild()}catch{}stamp();return n}
// WIP16's lexical placements() cannot be overridden directly; expose an Array named lastPlacements globally where sloppy-script lookup can resolve it.
function bridge(){const n=publish();try{window.lastPlacements=window.__muLivePreviewPlacements||[]}catch{};return n}
function run(){bridge();stamp();setTimeout(()=>{bridge();try{document.getElementById('muzUseLiveNom')?.click?.()}catch{}stamp()},60)}
for(const id of['muzUseLiveNom','generateBtn','generate','buildBtn','mlrGenerate'])document.getElementById(id)?.addEventListener('click',()=>{bridge();setTimeout(()=>{bridge();stamp()},120)},true);
try{fileInput?.addEventListener('change',()=>{setTimeout(bridge,100);setTimeout(stamp,250)},true)}catch{};
window.__muRefreshLiveNomPreview=refresh;bridge();stamp();
})();
