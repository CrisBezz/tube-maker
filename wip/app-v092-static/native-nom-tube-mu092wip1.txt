function parseGlbGroups(file){
 if(!file||!file.name?.toLowerCase().endsWith('.glb')){sourcePrimitiveGroups=[];return;}

   const dv=new DataView(buf); if(dv.getUint32(0,true)!==0x46546c67)return;

   while(off+8<=buf.byteLength){const len=dv.getUint32(off,true),type=dv.getUint32(off+4,true);off+=8;if(type===0x4e4f534a){json=JSON.parse(new TextDecoder().decode(new Uint8Array(buf,off,len)).replace(/\0+$/,''));break;}off+=len;}
   const groups=[];for(const mesh of json?.meshes||[])for(const prim of mesh.primitives||[])groups.push(prim?.extras?.nomad?.group??null);sourcePrimitiveGroups=groups;
 }catch(e){console.warn('MeshUtilz NOM export: source group parse failed',e);sourcePrimitiveGroups=[];}});
}

function readU64(dv,o){return Number(dv.getBigUint64(o,true))}
function writeU64(dv,o,v){dv.setBigUint64(o,BigInt(v),true)}
function parseNom(bytes){
 if(bytes.length<72||new TextDecoder().decode(bytes.slice(0,11))!=='Nomad Sculp'){
   // full magic is 12 bytes: "Nomad Sculpt"
 }
 const magic=new TextDecoder().decode(bytes.slice(0,12));
 if(magic!=='Nomad Sculpt')throw new Error('Not a Nomad Sculpt NOM file');
 const dv=new DataView(bytes.buffer,bytes.byteOffset,bytes.byteLength),jsonOffset=readU64(dv,24),jsonLength=readU64(dv,32),binOffset=readU64(dv,40),binLength=readU64(dv,48);
 if(jsonOffset+jsonLength>bytes.length||binOffset+binLength>bytes.length)throw new Error('Invalid NOM container offsets');
 const json=JSON.parse(new TextDecoder().decode(bytes.slice(jsonOffset,jsonOffset+jsonLength)));
 return{jsonOffset,jsonLength,binOffset,binLength,json};
}
function firstTubeMesh(project){return project?.meshes?.find(m=>m?.mesh_type==='tube'&&m?.config_tube)||null}

function qkey(p,t){return `${Math.round(p.x/t)},${Math.round(p.y/t)},${Math.round(p.z/t)}`}
function objGroup(obj,fallback){
 const vals=[obj?.userData?.nomad?.group,obj?.geometry?.userData?.nomad?.group,obj?.userData?.extras?.nomad?.group,obj?.geometry?.userData?.extras?.nomad?.group];
 for(const v of vals)if(v!==undefined&&v!==null)return v;return fallback;
}
function groupedTriangles(){
 if(!sourceRoot)return[];sourceRoot.updateMatrixWorld(true);const tris=[];let primitiveIndex=0;
 sourceRoot.traverse(obj=>{
   if(!obj.isMesh||!obj.geometry?.attributes?.position)return;
   const g=obj.geometry,p=g.getAttribute('position'),idx=g.index,groups=g.groups?.length?g.groups:[{start:0,count:idx?idx.count:p.count}];
   for(let gi=0;gi<groups.length;gi++){
     const gr=groups[gi],gid=objGroup(obj,sourcePrimitiveGroups[primitiveIndex+gi]??primitiveIndex+gi),start=gr.start||0,end=Math.min(start+(gr.count||0),idx?idx.count:p.count);
     for(let i=start;i+2<end;i+=3){const ids=[idx?idx.getX(i):i,idx?idx.getX(i+1):i+1,idx?idx.getX(i+2):i+2],v=ids.map(k=>new THREE.Vector3(p.getX(k),p.getY(k),p.getZ(k)).applyMatrix4(obj.matrixWorld));const n=v[1].clone().sub(v[0]).cross(v[2].clone().sub(v[0]));if(n.lengthSq()>1e-18)tris.push({v,normal:n.normalize(),group:gid});}
   }
   primitiveIndex+=Math.max(groups.length,1);
 });return tris;
}
function boundarySegments(){
 const tris=groupedTriangles();if(!tris.length)return[];const box=new THREE.Box3();tris.forEach(t=>t.v.forEach(p=>box.expandByPoint(p)));const tol=Math.max(box.getSize(new THREE.Vector3()).length(),1e-6)*1e-6,edges=new Map();
 const add=(a,b,t)=>{const ka=qkey(a,tol),kb=qkey(b,tol),key=ka<kb?`${ka}|${kb}`:`${kb}|${ka}`;let e=edges.get(key);if(!e){const f=ka<kb;e={ka:f?ka:kb,kb:f?kb:ka,a:(f?a:b).clone(),b:(f?b:a).clone(),refs:[]};edges.set(key,e)}e.refs.push(t)};
 for(const t of tris){add(t.v[0],t.v[1],t);add(t.v[1],t.v[2],t);add(t.v[2],t.v[0],t)}
 const out=[];for(const e of edges.values()){
   if(new Set(e.refs.map(r=>String(r.group))).size<2)continue;
   let n=new THREE.Vector3(),seed=e.refs[0].normal.clone();for(const r of e.refs){const rn=r.normal.clone();if(rn.dot(seed)<0)rn.negate();n.add(rn)}if(n.lengthSq()>1e-14)out.push({...e,normal:n.normalize()});
 }return out;
}
function traceChains(segments){
 const adj=new Map();segments.forEach((s,i)=>{for(const k of [s.ka,s.kb]){let a=adj.get(k);if(!a){a=[];adj.set(k,a)}a.push(i)}});const used=new Uint8Array(segments.length),chains=[];
 function walk(seed,key){const points=[],normals=[];let ei=seed,guard=0;while(ei!=null&&!used[ei]&&guard++<segments.length+3){const s=segments[ei];used[ei]=1;const f=s.ka===key,a=f?s.a:s.b,b=f?s.b:s.a,nextKey=f?s.kb:s.ka;if(!points.length)points.push(a.clone());points.push(b.clone());normals.push(s.normal.clone());const next=(adj.get(nextKey)||[]).find(j=>!used[j]);if(next==null)break;ei=next;key=nextKey;}const closed=points.length>2&&points[0].distanceToSquared(points[points.length-1])<1e-10;if(closed)points.pop();return{points,normals,closed};}
 for(const [key,list] of adj)if(list.length!==2)for(const ei of list)if(!used[ei])chains.push(walk(ei,key));for(let i=0;i<segments.length;i++)if(!used[i])chains.push(walk(i,segments[i].ka));return chains.filter(c=>c.points.length>=2);
}
function chainLength(c){let s=0;for(let i=0;i<c.points.length-1;i++)s+=c.points[i].distanceTo(c.points[i+1]);if(c.closed&&c.points.length>2)s+=c.points[c.points.length-1].distanceTo(c.points[0]);return s}
function pointSegDistance(p,a,b){const ab=b.clone().sub(a),d=ab.lengthSq();if(d<1e-18)return p.distanceTo(a);const t=Math.max(0,Math.min(1,p.clone().sub(a).dot(ab)/d));return p.distanceTo(a.clone().addScaledVector(ab,t))}
function rdp(points,tol){
 if(points.length<=2)return points.slice();let max=-1,idx=-1;for(let i=1;i<points.length-1;i++){const d=pointSegDistance(points[i],points[0],points[points.length-1]);if(d>max){max=d;idx=i}}
 if(max>tol){const a=rdp(points.slice(0,idx+1),tol),b=rdp(points.slice(idx),tol);return a.slice(0,-1).concat(b)}return[points[0],points[points.length-1]];
}
function simplifyClosed(points,detail){
 if(points.length<=8||detail>=99.5)return points.map(p=>p.clone());const box=new THREE.Box3().setFromPoints(points),diag=Math.max(box.getSize(new THREE.Vector3()).length(),1e-9),f=Math.max(0,Math.min(1,(100-detail)/100)),tol=diag*f*f*0.035;
 // Break the ring at the point farthest from point 0, simplify both arcs, then rejoin.
 let far=1,farD=0;for(let i=1;i<points.length;i++){const d=points[0].distanceToSquared(points[i]);if(d>farD){farD=d;far=i}}
 const a=rdp(points.slice(0,far+1),tol),b=rdp(points.slice(far).concat(points.slice(0,1)),tol);const out=a.slice(0,-1).concat(b.slice(0,-1));return out.length>=4?out:points.filter((_,i)=>i%Math.max(1,Math.floor(points.length/4))===0).slice(0,4);
}
function averageNormal(chain){const n=new THREE.Vector3();for(const x of chain.normals||[])n.add(x);if(n.lengthSq()<1e-12)n.set(0,1,0);return n.normalize()}
function initialFrameNormal(points,surfaceNormal){
 if(points.length<2)return[1,0,0];const tangent=points[1].clone().sub(points[0]).normalize();let n=surfaceNormal.clone();n.addScaledVector(tangent,-n.dot(tangent));if(n.lengthSq()<1e-12){const h=Math.abs(tangent.y)<.9?new THREE.Vector3(0,1,0):new THREE.Vector3(1,0,0);n.crossVectors(tangent,h)}return n.normalize().toArray();
}

function rebuildNom(templateBytes,project){
 const parsed=parseNom(templateBytes),encoder=new TextEncoder(),jsonBytes=encoder.encode(JSON.stringify(project,null,4)),newBinOffset=(parsed.jsonOffset+jsonBytes.length+7)&~7,bin=templateBytes.slice(parsed.binOffset,parsed.binOffset+parsed.binLength),size=newBinOffset+bin.length,out=new Uint8Array(size);

 const dv=new DataView(out.buffer);writeU64(dv,16,size);writeU64(dv,24,parsed.jsonOffset);writeU64(dv,32,jsonBytes.length);writeU64(dv,40,newBinOffset);writeU64(dv,48,bin.length);return out;
}
