function extract(){
 try{
  if(typeof resultRoot==='undefined'||!resultRoot)return[];
  resultRoot.updateMatrixWorld?.(true);
  const sets=[];resultRoot.traverse?.(o=>{if(o?.isInstancedMesh&&o.count>0)sets.push(o)});
  if(!sets.length&&Array.isArray(resultRoot.children))for(const o of resultRoot.children)if(o?.isInstancedMesh&&o.count>0)sets.push(o);
  if(!sets.length)return[];
  // Imported masters can contain several instanced mesh parts sharing the same placement transforms. Use the largest set once.
  sets.sort((a,b)=>b.count-a.count);const src=sets[0],m=new THREE.Matrix4(),wm=new THREE.Matrix4(),p=new THREE.Vector3(),q=new THREE.Quaternion(),s=new THREE.Vector3(),out=[];
  for(let i=0;i<src.count;i++){
   src.getMatrixAt(i,m);wm.multiplyMatrices(src.matrixWorld||new THREE.Matrix4(),m);wm.decompose(p,q,s);
   if(Math.abs(s.x)<1e-10||Math.abs(s.y)<1e-10||Math.abs(s.z)<1e-10)continue;
   out.push({position:p.clone(),quaternion:q.clone(),scale:s.clone(),matrix:wm.toArray()});
  }
  return out;
 }catch(e){console.warn('WIP20 authoritative transform extraction failed',e);return[]}
}

function u64(dv,o){return Number(dv.getBigUint64(o,true))}function w64(dv,o,v){dv.setBigUint64(o,BigInt(v),true)}
function parse(bytes){const dv=new DataView(bytes.buffer,bytes.byteOffset,bytes.byteLength),jo=u64(dv,24),jl=u64(dv,32),bo=u64(dv,40),bl=u64(dv,48);return{jo,bin:bytes.slice(bo,bo+bl),json:JSON.parse(new TextDecoder().decode(bytes.slice(jo,jo+jl)))}}
function patchNom(bytes){const t=publish();if(!t.length)return bytes;const p=parse(bytes),j=p.json;let nodes=(j.scene?.length===1&&Array.isArray(j.scene[0]?.children))?j.scene[0].children:(j.scene||[]);const n=Math.min(nodes.length,t.length);for(let i=0;i<n;i++)nodes[i].matrix=t[i].matrix.slice();if(nodes.length>t.length)nodes.splice(t.length);const jb=new TextEncoder().encode(JSON.stringify(j,null,4)),bo=(p.jo+jb.length+7)&~7,out=new Uint8Array(bo+p.bin.length);out.set(bytes.slice(0,p.jo));out.set(jb,p.jo);out.set(p.bin,bo);const dv=new DataView(out.buffer);w64(dv,16,out.length);w64(dv,24,p.jo);w64(dv,32,jb.length);w64(dv,40,bo);w64(dv,48,p.bin.length);return out}
