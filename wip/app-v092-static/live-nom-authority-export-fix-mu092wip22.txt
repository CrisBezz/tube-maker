
// MeshUtilz v0.9.2-wip.22 — repair Live NOM export authority after WIP21 button replacement
(function(){
const VERSION='0.9.2-wip.22',PRODUCT='MeshUtilz';
function stamp(){document.title=`${PRODUCT} v${VERSION}`;document.querySelectorAll('.version').forEach(x=>x.textContent=`v${VERSION}`);const av=document.querySelector('.about-version');if(av)av.textContent=`${PRODUCT} · Version ${VERSION}`;const bars=[...document.querySelectorAll('.mlr-statusbar,.mlr-wip8-statusbar')];bars.slice(1).forEach(x=>x.remove());if(bars[0]){const s=bars[0].querySelectorAll('span');if(s.length)s[s.length-1].textContent=`v${VERSION}`}}
function extract(){try{if(typeof resultRoot==='undefined'||!resultRoot)return[];resultRoot.updateMatrixWorld?.(true);const sets=[];resultRoot.traverse?.(o=>{if(o?.isInstancedMesh&&o.count>0)sets.push(o)});if(!sets.length)return[];sets.sort((a,b)=>b.count-a.count);const src=sets[0],m=new THREE.Matrix4(),wm=new THREE.Matrix4(),p=new THREE.Vector3(),q=new THREE.Quaternion(),s=new THREE.Vector3(),out=[];for(let i=0;i<src.count;i++){src.getMatrixAt(i,m);wm.multiplyMatrices(src.matrixWorld||new THREE.Matrix4(),m);wm.decompose(p,q,s);if(Math.abs(s.x)<1e-10||Math.abs(s.y)<1e-10||Math.abs(s.z)<1e-10)continue;out.push({position:p.clone(),quaternion:q.clone(),scale:s.clone(),matrix:wm.toArray()})}return out}catch(e){console.warn('WIP22 extraction failed',e);return[]}}
function publish(){const a=extract();if(a.length){window.__muLiveAuthoritativeTransforms=a;window.__muLivePreviewPlacements=a;window.lastPlacements=a;}return a}
function bind(){for(const id of['generateBtn','generate','buildBtn','mlrGenerate'])document.getElementById(id)?.addEventListener('click',()=>{setTimeout(()=>{const a=publish();try{window.__muRefreshLiveNomPreview?.()}catch{};if(a.length)setStatus?.(`WIP22: ${a.length} authoritative transforms captured for Live NOM export`);stamp()},420)},false);
 const exp=document.getElementById('muzExportLiveNom');if(exp)exp.addEventListener('click',(ev)=>{const a=publish();if(!a.length){ev.stopImmediatePropagation();alert('No generated instance transforms were found. Generate Imported Instance placements first, then export.');setStatus?.('Live NOM export: no generated InstanceMesh found');}else{setStatus?.(`Exporting ${a.length} linked Live NOM instances…`)}stamp()},true);
}
requestAnimationFrame(()=>{bind();publish();stamp();setTimeout(stamp,600)});
})();
