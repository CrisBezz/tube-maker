function collectFaces(){const out=[];if(typeof sourceRoot==='undefined'||!sourceRoot)return out;sourceRoot.updateMatrixWorld(true);sourceRoot.traverse(o=>{if(!o.isMesh||!o.geometry?.attributes?.position)return;const p=o.geometry.getAttribute('position'),ix=o.geometry.index,count=ix?ix.count:p.count,nm=new THREE.Matrix3().getNormalMatrix(o.matrixWorld);for(let i=0;i+2<count;i+=3){const ids=[ix?ix.getX(i):i,ix?ix.getX(i+1):i+1,ix?ix.getX(i+2):i+2],lv=ids.map(k=>new THREE.Vector3(p.getX(k),p.getY(k),p.getZ(k))),wv=lv.map(v=>v.clone().applyMatrix4(o.matrixWorld)),c=wv[0].clone().add(wv[1]).add(wv[2]).multiplyScalar(1/3);let n=lv[1].clone().sub(lv[0]).cross(lv[2].clone().sub(lv[0])).normalize().applyMatrix3(nm).normalize();const edges=[[wv[0],wv[1]],[wv[1],wv[2]],[wv[2],wv[0]]];edges.sort((a,b)=>b[0].distanceToSquared(b[1])-a[0].distanceToSquared(a[1]));let t=edges[0][1].clone().sub(edges[0][0]);t.addScaledVector(n,-t.dot(n));if(t.lengthSq()<1e-12){const h=Math.abs(n.y)<.9?new THREE.Vector3(0,1,0):new THREE.Vector3(1,0,0);t.crossVectors(h,n)}t.normalize();out.push({position:c,normal:n,tangent:t})}});return out}
function smoothed(records,i,amount){const base=records[i];if(amount<=0||records.length<2)return{normal:base.normal.clone(),tangent:base.tangent.clone()};const near=records.map((r,j)=>({r,j,d:r.position.distanceToSquared(base.position)})).sort((a,b)=>a.d-b.d).slice(0,12),an=new THREE.Vector3(),at=new THREE.Vector3();let ws=0;for(const q of near){const w=1/(Math.sqrt(q.d)+1e-6),n=q.r.normal.clone(),t=q.r.tangent.clone();if(n.dot(base.normal)<0)n.negate();if(t.dot(base.tangent)<0)t.negate();an.addScaledVector(n,w);at.addScaledVector(t,w);ws+=w}if(ws<=0||an.lengthSq()<1e-12)return{normal:base.normal.clone(),tangent:base.tangent.clone()};an.normalize();const n=base.normal.clone().lerp(an,amount).normalize();at.multiplyScalar(1/ws);at.addScaledVector(n,-at.dot(n));if(at.lengthSq()<1e-12)at.copy(base.tangent).addScaledVector(n,-base.tangent.dot(n));at.normalize();let t=base.tangent.clone();t.addScaledVector(n,-t.dot(n));if(t.lengthSq()<1e-12)t.copy(at);else t.normalize();t.lerp(at,amount);t.addScaledVector(n,-t.dot(n));if(t.lengthSq()<1e-12)t.copy(at);t.normalize();return{normal:n,tangent:t}}
function sourceSecondary(axis){const candidates=[new THREE.Vector3(1,0,0),new THREE.Vector3(0,1,0),new THREE.Vector3(0,0,1)].sort((a,b)=>Math.abs(a.dot(axis))-Math.abs(b.dot(axis)));const s=candidates[0].clone();s.addScaledVector(axis,-s.dot(axis));return s.normalize()}
