const ADAPTERS={
  BOX:{
    title:'Box dimensions',
    fields:[['size_x','Width'],['size_y','Height'],['size_z','Depth']],
    config:'config_box',
    map(p,source,target){return[p[0]*target.size_x/source.size_x,p[1]*target.size_y/source.size_y,p[2]*target.size_z/source.size_z]}
  },
  CYLINDER:{
    title:'Cylinder dimensions',
    fields:[['height','Length'],['radius_start','Bottom radius'],['radius_end','Top radius']],
    config:'config_cylinder',
    map(p,source,target){
      const yRatio=target.height/source.height,t=Math.max(0,Math.min(1,p[1]/source.height+.5));
      const oldRadius=source.radius_start+(source.radius_end-source.radius_start)*t;
      const newRadius=target.radius_start+(target.radius_end-target.radius_start)*t;
      const radial=oldRadius>1e-9?newRadius/oldRadius:1;
      return[p[0]*radial,p[1]*yRatio,p[2]*radial];
    }
  },
  TORUS:{
    title:'Torus dimensions',
    fields:[['radius_outer','Outer radius'],['radius_inner','Inner radius']],
    config:'config_torus',
    map(p,source,target){
      const rho=Math.hypot(p[0],p[2]),minorScale=target.radius_inner/source.radius_inner;
      const newRho=target.radius_outer+(rho-source.radius_outer)*minorScale;
      const radial=rho>1e-9?newRho/rho:1;
      return[p[0]*radial,p[1]*minorScale,p[2]*radial];
    }
  },
  TUBE:{
    title:'Tube dimensions',fields:[['length','Length'],['radius','Radius']],config:'config_tube',
    read(config){const points=config?.curve?.points||[],a=points[0]||[0,-.5,0],b=points.at(-1)||[0,.5,0];return{length:Math.hypot(b[0]-a[0],b[1]-a[1],b[2]-a[2])||1,radius:Number(config?.radiuses?.[0])||.5}},

    map(p,source,target){return[p[0]*target.radius/source.radius,p[1]*target.length/source.length,p[2]*target.radius/source.radius]}
  }

function mutateVertices(live,adapter,state){
  live.parsed.bin=state.sourceBin.slice();
  if(!state.fields.length)return;
  const field=live.mesh.vertices;
  if(field?.lz4)throw Error('RC2 parameter editing does not support compressed/LZ4 '+live.kind+' vertex caches');
  if(field?.type!=='f32vec3'||!Number.isFinite(field.count)||field.length<field.count*12)throw Error('RC2 parameter editing requires an uncompressed f32vec3 vertex cache');
  const source=new DataView(state.sourceBin.buffer,state.sourceBin.byteOffset,state.sourceBin.byteLength),out=new DataView(live.parsed.bin.buffer,live.parsed.bin.byteOffset,live.parsed.bin.byteLength);
  const base=field.offset,target=state.values,sourceValues=adapter.read?adapter.read(state.sourceConfig):state.sourceConfig;
  for(let i=0;i<field.count;i++){
    const o=base+i*12,p=[source.getFloat32(o,true),source.getFloat32(o+4,true),source.getFloat32(o+8,true)],q=adapter.map(p,sourceValues,target);
    out.setFloat32(o,q[0],true);out.setFloat32(o+4,q[1],true);out.setFloat32(o+8,q[2],true);
  }
}

function fieldBounds(live){
  const a=live.mesh.vertices,dv=new DataView(live.parsed.bin.buffer,live.parsed.bin.byteOffset,live.parsed.bin.byteLength),lo=[Infinity,Infinity,Infinity],hi=[-Infinity,-Infinity,-Infinity];

function close(a,b,t=3e-3){return Math.abs(a-b)<=t*Math.max(1,Math.abs(a),Math.abs(b))}
function validateLive(live){
  const adapter=ADAPTERS[live?.kind],state=live?.parameterState;if(!adapter||!state||!state.fields.length)return true;
  if(live.mesh.vertices?.lz4)throw Error('RC2 validation failed: edited compressed/LZ4 '+live.kind+' cache');
  const cfg=live.mesh[adapter.config],actual=adapter.read?adapter.read(cfg):cfg;for(const [key] of state.fields)if(!close(Number(actual[key]),Number(state.values[key]),1e-6))throw Error('RC2 validation failed: '+key+' does not match the parameter panel');
  const bounds=fieldBounds(live),v=state.values;
  if(live.kind==='BOX'&&(!close(bounds.size[0],v.size_x)||!close(bounds.size[1],v.size_y)||!close(bounds.size[2],v.size_z)))throw Error('RC2 validation failed: Box cache dimensions do not match its Nomad parameters');
  if(live.kind==='CYLINDER'&&!close(bounds.size[1],v.height))throw Error('RC2 validation failed: Cylinder cache length does not match its Nomad height');
  if(live.kind==='TORUS'&&(!close(bounds.size[1],v.radius_inner*2)||!close(bounds.size[0],(v.radius_outer+v.radius_inner)*2)))throw Error('RC2 validation failed: Torus cache dimensions do not match its Nomad radii');
  if(live.kind==='TUBE'&&live.internalFactory&&(!close(bounds.size[1],v.length)||!close(bounds.size[0],v.radius*2)))throw Error('RC2 validation failed: Tube cache dimensions do not match its Nomad length and radius');
  return true;
}
