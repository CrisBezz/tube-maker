function groupedNomadExtras(matrix, selected = false) {
  return nomadNodeExtras(matrix, selected);
}

buildNomadSharedNodeScene = function(instancedMesh) {
  const exportScene = new THREE.Scene();
  exportScene.name = 'NomadGroupedTorusInstances';

  const group = new THREE.Group();
  group.name = 'Generated Instances';
  group.matrixAutoUpdate = false;
  group.matrix.identity();
  group.userData = groupedNomadExtras(group.matrix, false);
  exportScene.add(group);

  const sharedGeometry = instancedMesh.geometry;
  const sharedMaterial = instancedMesh.material;
  const matrix = new THREE.Matrix4();
  let made = 0;

  for (let i = 0; i < instancedMesh.count; i++) {
    instancedMesh.getMatrixAt(i, matrix);
    const e = matrix.elements;
    const sx = Math.hypot(e[0], e[1], e[2]);
    const sy = Math.hypot(e[4], e[5], e[6]);
    const sz = Math.hypot(e[8], e[9], e[10]);
    if (sx < 1e-10 || sy < 1e-10 || sz < 1e-10) continue;

    const node = new THREE.Mesh(sharedGeometry, sharedMaterial);
    node.name = `Torus ${String(made + 1).padStart(3, '0')}`;
    node.matrixAutoUpdate = false;
    node.matrix.copy(matrix);
    node.userData = groupedNomadExtras(matrix, false);
    group.add(node);
    made++;
  }

  exportScene.userData = {
    meshLineReplacer: {
      version: GROUPED_WIP_VERSION,
      instanceMode: 'shared-gltf-nodes',
      hierarchy: 'grouped',
      groupName: group.name,
      instanceCount: made,
      target: 'Nomad Sculpt'
    }
  };
  return exportScene;
};

if (typeof buildCustomSharedNodeExport !== 'undefined') {
  buildCustomSharedNodeExport = function() {
    const exportScene = new THREE.Scene();
    exportScene.name = 'NomadGroupedImportedMasterInstances';

    const group = new THREE.Group();
    group.name = `${customMasterName || 'Imported Master'} Instances`;
    group.matrixAutoUpdate = false;
    group.matrix.identity();
    group.userData = groupedNomadExtras(group.matrix, false);
    exportScene.add(group);

    const matrix = new THREE.Matrix4();
    let childCount = 0;
    const instancedParts = resultRoot.children.filter((c) => c.isInstancedMesh);

    instancedParts.forEach((instanced, partIndex) => {
      for (let i = 0; i < instanced.count; i++) {
        instanced.getMatrixAt(i, matrix);
        const mesh = new THREE.Mesh(instanced.geometry, instanced.material);
        const baseName = instanced.name || customMasterName || `Master Part ${partIndex + 1}`;
        mesh.name = `${baseName} ${String(i + 1).padStart(3, '0')}`;
        mesh.matrixAutoUpdate = false;
        mesh.matrix.copy(matrix);
        mesh.userData = groupedNomadExtras(matrix, false);
        group.add(mesh);
        childCount++;
      }
    });

    exportScene.userData = {
      meshLineReplacer: {
        version: GROUPED_WIP_VERSION,
        instanceMode: 'shared-gltf-nodes',
        hierarchy: 'grouped',
        groupName: group.name,
        master: customMasterName,
        placementCount: resultRoot.userData?.customInstanceSet?.count || 0,
        meshNodeCount: childCount,
        target: 'Nomad Sculpt'
      }
    };
    return exportScene;
  };
}
