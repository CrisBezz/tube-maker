function addEdge(map, a, b) {
  const keyFor = (v) => `${v.x.toPrecision(9)},${v.y.toPrecision(9)},${v.z.toPrecision(9)}`;
  const ka = keyFor(a), kb = keyFor(b), key = ka < kb ? `${ka}|${kb}` : `${kb}|${ka}`;
  if (!map.has(key)) map.set(key, [a.clone(), b.clone()]);
}
function edgeIndexKey(a, b) { return a < b ? `${a}|${b}` : `${b}|${a}`; }
function prepareMergedGeometry(geometry) {
  const positionOnly = new THREE.BufferGeometry();
  positionOnly.setAttribute('position', geometry.getAttribute('position').clone());
  if (geometry.index) positionOnly.setIndex(geometry.index.clone());
  const working = mergeVertices(positionOnly, 1e-6);
  positionOnly.dispose();
  return working;
}
function orderQuad(unique1, shared0, unique2, shared1, pos) {
  const pts = [unique1, shared0, unique2, shared1].map((id) => new THREE.Vector3().fromBufferAttribute(pos, id));
  const centre = pts.reduce((acc, p) => acc.add(p), new THREE.Vector3()).multiplyScalar(1 / 4);
  const normal = pts[1].clone().sub(pts[0]).cross(pts[3].clone().sub(pts[0])).normalize();
  const u = pts[0].clone().sub(centre).normalize();
  const v = normal.clone().cross(u).normalize();
  const order = [0, 1, 2, 3].sort((ia, ib) => {
    const pa = pts[ia].clone().sub(centre), pb = pts[ib].clone().sub(centre);
    const aa = Math.atan2(pa.dot(v), pa.dot(u));
    const ab = Math.atan2(pb.dot(v), pb.dot(u));
    return aa - ab;
  });
  return order.map((i) => [unique1, shared0, unique2, shared1][i]);
}
function reconstructQuadData(geometry) {
  const working = prepareMergedGeometry(geometry);
  const pos = working.getAttribute('position');
  const index = working.index;
  if (!index || !pos) { working.dispose(); return { edges: [], faces: [], pairs: 0, unmatched: 0 }; }
  const triangles = [];
  const adjacency = new Map();
  for (let i = 0; i < index.count; i += 3) {
    const ids = [index.getX(i), index.getX(i + 1), index.getX(i + 2)];
    const points = ids.map((id) => new THREE.Vector3().fromBufferAttribute(pos, id));
    const normal = points[1].clone().sub(points[0]).cross(points[2].clone().sub(points[0])).normalize();
    const triId = triangles.length;
    triangles.push({ ids, points, normal });
    for (let e = 0; e < 3; e++) {
      const a = ids[e], b = ids[(e + 1) % 3], key = edgeIndexKey(a, b);
      if (!adjacency.has(key)) adjacency.set(key, []);
      adjacency.get(key).push({ triId, a, b });
    }
  }
  const tolerance = THREE.MathUtils.degToRad(Number(quadAngle.value));
  const candidates = [];
  for (const [sharedKey, refs] of adjacency) {
    if (refs.length !== 2) continue;
    const t1 = triangles[refs[0].triId], t2 = triangles[refs[1].triId];
    const normalAngle = Math.acos(THREE.MathUtils.clamp(t1.normal.dot(t2.normal), -1, 1));
    if (normalAngle > tolerance) continue;
    const shared = sharedKey.split('|').map(Number);
    const unique1 = t1.ids.find((id) => !shared.includes(id));
    const unique2 = t2.ids.find((id) => !shared.includes(id));
    if (unique1 == null || unique2 == null || unique1 === unique2) continue;
    const p0 = new THREE.Vector3().fromBufferAttribute(pos, shared[0]);
    const p1 = new THREE.Vector3().fromBufferAttribute(pos, shared[1]);
    const pu = new THREE.Vector3().fromBufferAttribute(pos, unique1);
    const pv = new THREE.Vector3().fromBufferAttribute(pos, unique2);
    const sharedLength = p0.distanceTo(p1);
    const alternateDiagonal = pu.distanceTo(pv);
    const balance = Math.abs(sharedLength - alternateDiagonal) / Math.max(sharedLength, alternateDiagonal, 1e-9);
    candidates.push({ triA: refs[0].triId, triB: refs[1].triId, shared, unique1, unique2, score: normalAngle + balance * 0.05 });
  }
  candidates.sort((a, b) => a.score - b.score);
  const matched = new Set();
  const edgeMap = new Map();
  const faces = [];
  let pairs = 0;
  const addById = (a, b) => {
    const va = new THREE.Vector3().fromBufferAttribute(pos, a);
    const vb = new THREE.Vector3().fromBufferAttribute(pos, b);
    addEdge(edgeMap, va, vb);
  };
  for (const c of candidates) {
    if (matched.has(c.triA) || matched.has(c.triB)) continue;
    matched.add(c.triA); matched.add(c.triB); pairs++;
    addById(c.unique1, c.shared[0]); addById(c.unique1, c.shared[1]); addById(c.unique2, c.shared[0]); addById(c.unique2, c.shared[1]);
    const ordered = orderQuad(c.unique1, c.shared[0], c.unique2, c.shared[1], pos);
    faces.push(ordered.map((id) => new THREE.Vector3().fromBufferAttribute(pos, id)));
  }
  if (unmatched.value === 'keep') {
    triangles.forEach((tri, triId) => {
      if (matched.has(triId)) return;
      addById(tri.ids[0], tri.ids[1]); addById(tri.ids[1], tri.ids[2]); addById(tri.ids[2], tri.ids[0]);
      faces.push(tri.ids.map((id) => new THREE.Vector3().fromBufferAttribute(pos, id)));
    });
  }
  const unmatchedCount = triangles.length - matched.size;
  working.dispose();
  return { edges: [...edgeMap.values()], faces, pairs, unmatched: unmatchedCount };
}
function collectEdges() {
  if (!sourceRoot) return { edges: [], info: '' };
  if (edgeMode.value === 'objQuads') {
    const filtered = (originalObjQuadEdges || []).filter(([a, b]) => a.distanceTo(b) >= Number(minLength.value || 0));
    return { edges: filtered, info: `${filtered.length.toLocaleString()} original OBJ quad edges` };
  }
  const edges = new Map();
  let quadPairs = 0, unmatchedTriangles = 0;
  sourceRoot.traverse((obj) => {
    if (!obj.isMesh) return;
    if (edgeMode.value === 'quads') {
      const result = reconstructQuadData(obj.geometry);
      quadPairs += result.pairs; unmatchedTriangles += result.unmatched;
      for (const [a, b] of result.edges) if (a.distanceTo(b) >= Number(minLength.value || 0)) addEdge(edges, a, b);
      return;
    }
    const edgeGeo = edgeMode.value === 'sharp' ? new THREE.EdgesGeometry(obj.geometry, Number(angle.value)) : new THREE.WireframeGeometry(obj.geometry);
    const p = edgeGeo.getAttribute('position');
    for (let i = 0; i < p.count; i += 2) {
      const a = new THREE.Vector3().fromBufferAttribute(p, i);
      const b = new THREE.Vector3().fromBufferAttribute(p, i + 1);
      if (a.distanceTo(b) >= Number(minLength.value || 0)) addEdge(edges, a, b);
    }
    edgeGeo.dispose();
  });
  const info = edgeMode.value === 'quads' ? `${quadPairs.toLocaleString()} reconstructed quads • ${unmatchedTriangles.toLocaleString()} unmatched triangles` : '';
  return { edges: [...edges.values()], info };
}
function collectFaces() {
  if (!sourceRoot) return { faces: [], info: '' };
  if (edgeMode.value === 'objQuads') {
    const faces = (originalObjQuadFaces || []).map((face) => face.map((p) => p.clone()));
    return { faces, info: `${faces.length.toLocaleString()} original OBJ quad faces` };
  }
  const faces = [];
  let quadPairs = 0, unmatchedTriangles = 0;
  sourceRoot.traverse((obj) => {
    if (!obj.isMesh) return;
    if (edgeMode.value === 'quads') {
      const result = reconstructQuadData(obj.geometry);
      quadPairs += result.pairs; unmatchedTriangles += result.unmatched;
      faces.push(...result.faces.map((face) => face.map((p) => p.clone())));
      return;
    }
    const working = prepareMergedGeometry(obj.geometry);
    const pos = working.getAttribute('position');
    const idx = working.index;
    if (idx && pos) {
      for (let i = 0; i < idx.count; i += 3) {
        const tri = [idx.getX(i), idx.getX(i + 1), idx.getX(i + 2)].map((id) => new THREE.Vector3().fromBufferAttribute(pos, id));
        const e0 = tri[0].distanceTo(tri[1]), e1 = tri[1].distanceTo(tri[2]), e2 = tri[2].distanceTo(tri[0]);
        if (Math.min(e0, e1, e2) >= Number(minLength.value || 0)) faces.push(tri);
      }
    }
    working.dispose();
  });
  const info = edgeMode.value === 'quads' ? `${quadPairs.toLocaleString()} reconstructed quads • ${unmatchedTriangles.toLocaleString()} unmatched triangles` : `${faces.length.toLocaleString()} source faces`;
  return { faces, info };
}
function applySymmetryToEdges(edges) {
  if (!symmetry.checked || !sourceRoot) return { edges, mirrored: false, centre: null };
  const axis = symmetryAxis.value;
  const axisIndex = axis === 'x' ? 0 : axis === 'y' ? 1 : 2;
  const centre = symmetryCentre();
  const c = centre.getComponent(axisIndex);
  const box = sourceBounds();
  const span = box.getSize(new THREE.Vector3()).getComponent(axisIndex);
  const tolerance = Math.max(span * 1e-7, 1e-7);
  const positive = symmetrySide.value === 'positive';
  const kept = new Map();
  const add = (a,b) => addEdge(kept,a,b);
  for (const [a0,b0] of edges) {
    const midpoint = (a0.getComponent(axisIndex) + b0.getComponent(axisIndex)) * 0.5;
    if ((positive && midpoint < c - tolerance) || (!positive && midpoint > c + tolerance)) continue;
