
// v0.8.9-wip.2 — expand instance density from -5 to +5
(function(){
  const VERSION='0.8.9-wip.2';
  document.title=`Mesh Line Replacer v${VERSION}`;
  const v=document.querySelector('.version'); if(v)v.textContent=`v${VERSION}`;
  const av=document.querySelector('.about-version'); if(av)av.textContent=`Version ${VERSION}`;
  for(const input of [document.getElementById('torusDensity'),document.getElementById('customDensity')]){
    if(!input)continue; input.min='-5'; input.max='5'; input.step='1';
    const out=document.getElementById(input.id+'Out');
    const update=()=>{const n=Number(input.value); if(out)out.value=n>0?`+${n}`:`${n}`;};
    input.addEventListener('input',update); update();
  }
  function edgeNodesV2(faces,subdivisions){
    const map=new Map();
    for(const face of faces){
      const normal=polygonNormal(face); if(!normal)continue;
      const area=Math.max(faceAreaForNodeNormal(face),1e-12);
      for(let i=0;i<face.length;i++){
        const a=face[i],b=face[(i+1)%face.length],ka=vertexKey(a),kb=vertexKey(b),key=ka<kb?`${ka}|${kb}`:`${kb}|${ka}`;
        let rec=map.get(key);
        if(!rec){const edge=b.clone().sub(a);rec={a:a.clone(),b:b.clone(),tangent:edge,length:edge.length(),normalSum:new THREE.Vector3(),faces:0};map.set(key,rec);}
        rec.normalSum.addScaledVector(normal,area); rec.faces++;
      }
    }
    const out=[];
    for(const rec of map.values()){
      if(rec.length<=1e-9||rec.normalSum.lengthSq()<1e-14)continue;
      const normal=rec.normalSum.clone().normalize();
      for(let s=1;s<=subdivisions;s++){
        const t=s/(subdivisions+1);
        out.push({point:rec.a.clone().lerp(rec.b,t),normal:normal.clone(),tangent:rec.tangent.clone(),localRadius:rec.length/(subdivisions+1)*0.5,faceCount:rec.faces});
      }
    }
    return out;
  }
  densityRecordsForFaces=function(faces,placementMode,level){
    const base=placementMode==='node'?buildSharedNodePlacements(faces).map(p=>({kind:'node',source:p})):faces.map(f=>({kind:'face',source:f}));
    if(level<0){const ratios={'-1':0.8,'-2':0.6,'-3':0.4,'-4':0.25,'-5':0.125};return spatialThinDensity(base,ratios[String(level)]||0.5);}
    if(level===0)return base;
    const records=base.slice();
    edgeNodesV2(faces,Math.max(1,level)).forEach(p=>records.push({kind:'node',source:p}));
    if(level>=2){
      if(placementMode==='node')faces.forEach(f=>records.push({kind:'face',source:f}));
      else buildSharedNodePlacements(faces).forEach(p=>records.push({kind:'node',source:p}));
    }
    return dedupeDensityRecords(records);
  };
  if(typeof DENSITY_WIP_VERSION!=='undefined'){
    const body=document.querySelector('.about-body');
    const lists=body?.querySelectorAll('ul');
    if(lists?.length>1&&!body.textContent.includes('v0.8.9-wip.2')){
      const li=document.createElement('li');li.innerHTML='<strong>v0.8.9-wip.2</strong> — Instance density expanded to ten divisions (−5 to +5)';lists[1].prepend(li);
    }
  }
})();
