function addDensitySlider(afterControl, id) {
  if (!afterControl || document.getElementById(id)) return document.getElementById(id);
  const wrap = document.createElement('div');
  wrap.className = 'control';
  wrap.innerHTML = `<label for="${id}">Instance density</label><output id="${id}Out">0</output><input id="${id}" type="range" min="-2" max="2" value="0" step="1" />`;
  afterControl.insertAdjacentElement('afterend', wrap);
  const input = wrap.querySelector('input');
  const output = wrap.querySelector('output');
  const update = () => { const v = Number(input.value); output.value = v > 0 ? `+${v}` : `${v}`; };
  input.addEventListener('input', update); update();
  return input;
}

function densityFaceCentre(face) {
  const p = new THREE.Vector3();
  face.forEach((v) => p.add(v));
  return p.multiplyScalar(1 / Math.max(face.length, 1));
}
function densityFaceNode(face) {
  if (!face || face.length < 3) return null;
  const point = densityFaceCentre(face);
  const normal = polygonNormal(face);
  if (!normal) return null;
  let minEdge = Infinity, tangent = null;
  for (let i = 0; i < face.length; i++) {
    const edge = face[(i + 1) % face.length].clone().sub(face[i]);
    const len = edge.length();
    if (len > 1e-9) {
      minEdge = Math.min(minEdge, len);
      if (!tangent || edge.lengthSq() > tangent.lengthSq()) tangent = edge;
    }
  }
  let inRadius = Infinity;
  for (let i = 0; i < face.length; i++) inRadius = Math.min(inRadius, pointSegmentDistance3D(point, face[i], face[(i + 1) % face.length]));
  return { point, normal: normal.clone(), tangent: tangent || new THREE.Vector3(1,0,0), localRadius: Number.isFinite(inRadius) && inRadius > 1e-9 ? inRadius : (Number.isFinite(minEdge) ? minEdge * 0.5 : 1), faceCount: 1 };
}
function densityEdgeNodes(faces) {
  const map = new Map();
  for (const face of faces) {
    const normal = polygonNormal(face);
    if (!normal) continue;
    const area = Math.max(faceAreaForNodeNormal(face), 1e-12);
    for (let i = 0; i < face.length; i++) {
      const a = face[i], b = face[(i + 1) % face.length];
      const ka = vertexKey(a), kb = vertexKey(b);
      const key = ka < kb ? `${ka}|${kb}` : `${kb}|${ka}`;
      let rec = map.get(key);
      if (!rec) {
        const edge = b.clone().sub(a);
        rec = { a: a.clone(), b: b.clone(), tangent: edge, length: edge.length(), normalSum: new THREE.Vector3(), faces: 0 };
        map.set(key, rec);
      }
      rec.normalSum.addScaledVector(normal, area); rec.faces++;
    }
  }
  const out = [];
  for (const rec of map.values()) {
    if (rec.length <= 1e-9 || rec.normalSum.lengthSq() < 1e-14) continue;
    out.push({
      point: rec.a.clone().add(rec.b).multiplyScalar(0.5),
      normal: rec.normalSum.normalize(),
      tangent: rec.tangent.clone(),
      localRadius: rec.length * 0.5,
      faceCount: rec.faces
    });
  }
  return out;
}
function densityRecordPoint(rec) { return rec.kind === 'face' ? densityFaceCentre(rec.source) : rec.source.point; }
function densityRecordsForFaces(faces, placementMode, level) {
  const base = placementMode === 'node'
    ? buildSharedNodePlacements(faces).map((p) => ({ kind: 'node', source: p }))
    : faces.map((f) => ({ kind: 'face', source: f }));
  if (level < 0) return spatialThinDensity(base, level === -1 ? 0.5 : 0.25);
  if (level === 0) return base;

  const records = base.slice();
  densityEdgeNodes(faces).forEach((p) => records.push({ kind: 'node', source: p }));
  if (level >= 2) {
    if (placementMode === 'node') faces.forEach((f) => records.push({ kind: 'face', source: f }));
    else buildSharedNodePlacements(faces).forEach((p) => records.push({ kind: 'node', source: p }));
  }
  return dedupeDensityRecords(records);
}
function dedupeDensityRecords(records) {
  if (!records.length) return records;
  const box = new THREE.Box3(); records.forEach((r) => box.expandByPoint(densityRecordPoint(r)));
  const diag = Math.max(box.getSize(new THREE.Vector3()).length(), 1e-8);
  const tol = diag * 1e-8;
  const seen = new Set(), out = [];
  for (const rec of records) {
    const p = densityRecordPoint(rec);
    const key = `${Math.round(p.x/tol)},${Math.round(p.y/tol)},${Math.round(p.z/tol)}`;
    if (seen.has(key)) continue;
    seen.add(key); out.push(rec);
  }
  return out;
}
function spatialThinDensity(records, ratio) {
  if (records.length < 3 || ratio >= 0.999) return records.slice();
  const target = Math.max(1, Math.round(records.length * ratio));
  if (target >= records.length) return records.slice();
  const points = records.map(densityRecordPoint);
  const box = new THREE.Box3(); points.forEach((p) => box.expandByPoint(p));
  const min = box.min.clone();
  const diag = Math.max(box.getSize(new THREE.Vector3()).length(), 1e-9);
  function choose(cell) {
    const cells = new Map();
    for (let i = 0; i < records.length; i++) {
      const p = points[i];
      const key = `${Math.floor((p.x-min.x)/cell)},${Math.floor((p.y-min.y)/cell)},${Math.floor((p.z-min.z)/cell)}`;
      if (!cells.has(key)) cells.set(key, i);
    }
    return [...cells.entries()].sort((a,b)=>a[0].localeCompare(b[0])).map(([,i])=>records[i]);
  }
  let lo = diag / 1e6, hi = diag * 2, best = records.slice(), bestErr = Math.abs(records.length-target);
  for (let iter=0; iter<20; iter++) {
    const mid=(lo+hi)*0.5, chosen=choose(mid), err=Math.abs(chosen.length-target);
    if(err<bestErr){best=chosen;bestErr=err;}
    if(chosen.length>target) lo=mid; else hi=mid;
  }
  if(best.length>target) best=best.slice(0,target);
  return best;
}
function densityTorusData(rec) { return rec.kind === 'face' ? torusInstanceDataForFace(rec.source) : torusInstanceDataForNode(rec.source); }
function densityCustomMatrix(rec) { return rec.kind === 'face' ? customPlacementMatrixForFace(rec.source) : customPlacementMatrixForNode(rec.source); }
function densityLabel(level) { return level > 0 ? `+${level}` : `${level}`; }

generateFaceTorusResult = async function() {
  const facesResult = collectFaces();
  const orientedFaces = orientTorusFaces(facesResult.faces);
  const symmetryResult = applySymmetryToFaces(orientedFaces);
  const faces = symmetryResult.faces;
  if (!faces.length) { setStatus('No usable source faces were found for torus replacement'); return; }
  const placementMode = $('#torusPlacementMode')?.value === 'node' ? 'node' : 'face';
  const level = Number(torusDensity?.value || 0);
  const records = densityRecordsForFaces(faces, placementMode, level);
  if (!records.length) { setStatus('No usable torus placements were found'); return; }
  if (records.length > 100000 && !confirm(`This density will create ${records.length.toLocaleString()} torus instances. Continue?`)) { setStatus('Generation cancelled'); return; }
  const valid = [];
  for (const rec of records) { const data = densityTorusData(rec); if (data) valid.push({ rec, data }); }
  if (!valid.length) { setStatus('No valid torus placements could be generated'); return; }
  setStatus(`Creating ${valid.length.toLocaleString()} torus instances • density ${densityLabel(level)}…`);
  await new Promise((resolve) => requestAnimationFrame(resolve));
  disposeObject(resultRoot);
  resultRoot = new THREE.Group();
  resultRoot.name = placementMode === 'node' ? 'NodeTorusInstanceSet' : 'FaceTorusInstanceSet';
  const first = valid[0].data;
  const masterGeometry = new THREE.TorusGeometry(first.masterMajor, first.masterTube, Number(torusTubeSegments.value), Number(torusRingSegments.value));
  masterGeometry.name = 'MasterTorusGeometry';
  const material = new THREE.MeshStandardMaterial({ color: 0xd7a6f4, roughness: 0.55, metalness: 0.06, flatShading: torusFlatShading.checked });
  material.name = 'MasterTorusMaterial';
  const instanced = new THREE.InstancedMesh(masterGeometry, material, valid.length);
  instanced.name = placementMode === 'node' ? 'SharedNodeTorusInstances' : 'FaceTorusInstances';
  instanced.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
  valid.forEach(({data},i)=>instanced.setMatrixAt(i,data.matrix));
  instanced.instanceMatrix.needsUpdate = true;
  resultRoot.add(instanced); scene.add(resultRoot);
  sourceRoot.visible = showSource.checked; exportBtn.disabled = false;
  const masterFaces = masterGeometry.index ? masterGeometry.index.count/3 : masterGeometry.getAttribute('position').count/3;
  const label = placementMode === 'node' ? 'shared face nodes' : 'face centres';
  setStatus(`Generated ${valid.length.toLocaleString()} shared torus instances • ${label} • density ${densityLabel(level)}${level>0?' • topology-derived additions':''}${level<0?' • spatially thinned':''}${symmetryResult.mirrored?` • mirrored across ${symmetryAxis.value.toUpperCase()} centre plane`:''}`);
  setOutputMetrics({ faces: Math.round(masterFaces*valid.length), holes: valid.length, exported:false });
};
