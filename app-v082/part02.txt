function disposeObject(root) {
  if (!root) return;
  root.traverse((obj) => {
    if (obj.geometry) obj.geometry.dispose?.();
    const mat = obj.material;
    if (Array.isArray(mat)) mat.forEach((m) => m.dispose?.()); else mat?.dispose?.();
  });
  root.removeFromParent();
}
function fitCamera(object) {
  const box = new THREE.Box3().setFromObject(object);
  const sphere = box.getBoundingSphere(new THREE.Sphere());
  if (!Number.isFinite(sphere.radius) || sphere.radius <= 0) return;
  controls.target.copy(sphere.center);
  camera.near = Math.max(sphere.radius / 1000, 0.001);
  camera.far = Math.max(sphere.radius * 100, 100);
  camera.position.copy(sphere.center).add(new THREE.Vector3(1.5, 1.1, 1.5).normalize().multiplyScalar(sphere.radius * 3));
  camera.updateProjectionMatrix();
  controls.update();
  grid.scale.setScalar(Math.max(sphere.radius, 1));
  grid.position.y = box.min.y;
}
function normaliseImported(root) {
  const group = new THREE.Group();
  root.updateMatrixWorld(true);
  root.traverse((obj) => {
    if (!obj.isMesh || !obj.geometry) return;
    const geometry = obj.geometry.clone();
    geometry.applyMatrix4(obj.matrixWorld);
    if (!geometry.attributes.normal) geometry.computeVertexNormals();
    const material = new THREE.MeshStandardMaterial({ color: 0x9ba5b4, roughness: 0.75, metalness: 0.05, side: THREE.DoubleSide, transparent: true, opacity: 0.42 });
    group.add(new THREE.Mesh(geometry, material));
  });
  return group;
}
