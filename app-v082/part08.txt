face frame.
  const qLocalTilt = qTiltY.clone().multiply(qTiltX);
  const qFinal = qFrame.clone().multiply(qLocalTilt);
  geometry.applyQuaternion(qFinal);
  const offset = Number(torusNormalOffset.value || 0);
  const c = centre.clone().addScaledVector(normal, offset);
  geometry.translate(c.x, c.y, c.z);
  return geometry;
}
async function generateFaceTorusResult() {
  const facesResult = collectFaces();
  const orientedFaces = orientTorusFaces(facesResult.faces);
  const symmetryResult = applySymmetryToFaces(orientedFaces);
  const faces = symmetryResult.faces;
  if (!faces.length) { setStatus('No usable source faces were found for torus replacement'); return; }
  if (faces.length > 25000 && !confirm(`This model has ${faces.length.toLocaleString()} faces. One torus per face may be heavy on iPad memory. Continue?`)) { setStatus('Generation cancelled'); return; }
  setStatus(`Generating ${faces.length.toLocaleString()} face-aligned tori…`);
  await new Promise((resolve) => requestAnimationFrame(resolve));
  disposeObject(resultRoot);
  resultRoot = new THREE.Group();
  const material = new THREE.MeshStandardMaterial({ color: 0xd7a6f4, roughness: 0.55, metalness: 0.06, flatShading: torusFlatShading.checked });
  const batchSize = 300;
  let made = 0, skipped = 0, outputFaces = 0;
  for (let start = 0; start < faces.length; start += batchSize) {
    const geometries = [];
    for (const face of faces.slice(start, start + batchSize)) {
      const g = torusGeometryForFace(face);
      if (!g) { skipped++; continue; }
      outputFaces += g.index ? g.index.count / 3 : g.getAttribute('position').count / 3;
      geometries.push(g); made++;
    }
    if (geometries.length) {
      const merged = mergeGeometries(geometries, false);
      geometries.forEach((g) => g.dispose());
      if (merged) resultRoot.add(new THREE.Mesh(merged, material));
    }
    if (start % 1500 === 0) {
      setStatus(`Generating tori… ${Math.min(start + batchSize, faces.length).toLocaleString()} / ${faces.length.toLocaleString()}`);
      await new Promise((resolve) => requestAnimationFrame(resolve));
    }
  }
  if (!made) { setStatus('No valid tori could be generated from these faces'); return; }
  scene.add(resultRoot);
  sourceRoot.visible = showSource.checked;
  exportBtn.disabled = false;
  setStatus(`Generated ${made.toLocaleString()} face-aligned tori${skipped ? ` • ${skipped.toLocaleString()} degenerate faces skipped` : ''}${symmetryResult.mirrored ? ` • mirrored across ${symmetryAxis.value.toUpperCase()} centre plane` : ''}${facesResult.info ? ` • ${facesResult.info}` : ''}`);
  setOutputMetrics({ faces: Math.round(outputFaces), holes: made, exported: false });
}
async function generateResult() {
  if (!sourceRoot) return;
  generateBtn.disabled = true;
  try {
    if (replaceMode.value === 'surfaceWeb') await generateSurfaceWebResult();
    else if (replaceMode.value === 'solidWeb') await generateSolidWebResult();
    else if (replaceMode.value === 'faceTorus') await generateFaceTorusResult();
    else await generateTubeResult();
  } catch (error) {
    console.error(error);
    setStatus(`Generation failed: ${error.message || error}`);
  } finally {
    generateBtn.disabled = false;
  }
}
function exportGlb() {
  if (!resultRoot) return;
  exportBtn.disabled = true;
  setStatus('Preparing GLB…');
  const suffix = replaceMode.value === 'surfaceWeb' ? 'surface-web' : replaceMode.value === 'solidWeb' ? 'solid-surface-web' : replaceMode.value === 'faceTorus' ? 'face-tori' : 'tubes';
  new GLTFExporter().parse(resultRoot, (result) => {
    const blob = new Blob([result], { type: 'model/gltf-binary' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${sourceName}-${suffix}.glb`;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 3000);
    setStatus('GLB exported');
    setOutputMetrics({ exported: true });
    exportBtn.disabled = false;
  }, (error) => {
    console.error(error);
    setStatus(`Export failed: ${error.message || error}`);
    exportBtn.disabled = false;
  }, { binary: true, onlyVisible: true });
}

}
